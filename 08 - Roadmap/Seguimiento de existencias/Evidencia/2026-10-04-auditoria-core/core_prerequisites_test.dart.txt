void main() {
  for (final mode in [AppMode.standalone, AppMode.serverSync]) {
    group(mode.name, () {
    late AppDatabase db;
    late DriftLocalEventStore events;
    late DriftProductoProjectionStore products;
    late DriftInventoryProjectionStore inventory;
    late DriftVariantInventoryMemoryStore memory;
    late DriftVariantInventoryTrackingStore tracking;
    late ProductoCommandService commands;
    setUp(() {
      db = AppDatabase.forTesting(NativeDatabase.memory());
      products = DriftProductoProjectionStore(productoDao: ProductoDao(db));
      inventory = DriftInventoryProjectionStore(inventoryDao: InventoryDao(db), unitDao: UnitDao(db));
      memory = DriftVariantInventoryMemoryStore(dao: VariantInventoryMemoryDao(db));
      tracking = DriftVariantInventoryTrackingStore(inventoryDao: InventoryDao(db), productoDao: ProductoDao(db));
      events = DriftLocalEventStore(
        db: db, eventDao: EventDao(db), eventRefDao: EventRefDao(db),
        eventProcessor: EventProcessor(handlers: {
          ...inventoryEventHandlers(InventoryEventHandler(inventory)),
          ...productoEventHandlers(ProductoEventHandler(products, inventoryProjectionStore: inventory)),
        }),
        appConfigController: AppConfigController(AppConfig.initial.copyWith(mode: mode, setupCompleted: true)),
      );
      commands = ProductoCommandService(
        eventStore: events,
        commandContext: const LocalCommandContext(deviceId: 'audit', userId: 'audit'),
        categoriaProjectionStore: DriftCategoriaProjectionStore(categoriaDao: CategoriaDao(db)),
        productoProjectionStore: products,
        inventoryProjectionStore: inventory,
        variantInventoryMemoryStore: memory,
        variantInventoryTrackingStore: tracking,
        syncedEventHistory: DriftSyncPersistence(db: db, eventDao: EventDao(db), eventRefDao: EventRefDao(db), syncCheckpointDao: SyncCheckpointDao(db)),
        unidadInventarioRepository: UnidadInventarioRepositoryImpl(unitDao: UnitDao(db)),
      );
    });
    tearDown(() => db.close());
    Future<void> save(bool direct) async {
      final product = (await db.select(db.products).get()).single;
      final variant = (await db.select(db.productVariants).get()).single;
      await commands.actualizarArticulo(
        productId: product.id, baseEventId: product.lastEventId!, variantIds: [variant.id],
        command: CrearArticuloCommand.conVariantes(nombre: product.name, variantes: [
          CrearArticuloVarianteCommand(nombre: variant.name, precioVenta: '25', costoEstandar: null,
            inventoryUnitId: direct ? InventoryUnitIds.piece : null),
        ]),
      );
    }
    test('${mode.name}: alta por handlers establece memoria durable', () async {
      await events.appendAndApplyBatchAtomically(_entries(productInventoryItemId: _inventoryItemId));
      expect(await db.select(db.variantInventoryMemory).get(), hasLength(1));
    });
    test('${mode.name}: SEG-05 legado conserva ID al guardar apagado y reactivar', () async {
      await events.appendAndApplyBatchAtomically(_entries(productInventoryItemId: _inventoryItemId));
      await save(false);
      await save(true);
      expect((await db.select(db.productVariants).get()).single.inventoryItemId, _inventoryItemId);
      expect(await db.select(db.inventoryItems).get(), hasLength(1));
    });
    if (mode == AppMode.standalone) {
      test('standalone: SEG-06 descarta autogenerado vacío después de desvincular', () async {
        await commands.crearArticulo(const CrearArticuloCommand.conVariantes(nombre: 'Nuevo', variantes: [
          CrearArticuloVarianteCommand(nombre: null, precioVenta: '25', costoEstandar: null, inventoryUnitId: InventoryUnitIds.piece),
        ]));
        expect(await db.select(db.inventoryMovements).get(), isEmpty);
        await save(false);
        expect(await db.select(db.inventoryItems).get(), isEmpty);
        expect(await db.select(db.inventoryItemDiscards).get(), hasLength(1));
      });
    }
    test('${mode.name}: SEG-22 catálogo legado con historia incompleta requiere selección', () async {
      await events.appendAndApplyBatchAtomically(_entries(productInventoryItemId: _inventoryItemId));
      await save(false);
      // Simula un catálogo legado que llegó sin historia acreditable.
      await db.delete(db.events).go();
      final result = await InventoryResourceResolver(memoryStore: memory, trackingStore: tracking, inventoryStore: inventory).resolve(
        variantId: '10000000-0000-4000-8000-000000000002', currentInventoryItemId: null,
        requiredUnitId: InventoryUnitIds.piece, saleConfiguration: const UnitSaleConfiguration(),
      );
      expect(result, isA<RecursoSeleccionRequeridaResultado>());
    });
    if (mode == AppMode.serverSync) {
      test('serverSync: SEG-15 rollback restaura ausencia exacta de memoria', () async {
        await events.appendAndApplyBatchAtomically(_entries(productInventoryItemId: _inventoryItemId));
        final product = (await db.select(db.products).get()).single;
        final variant = (await db.select(db.productVariants).get()).single;
        const edit = '40000000-0000-4000-8000-000000000088';
        final dao = ProductoDao(db);
        await dao.guardarRespaldoActualizacion(edit, product.id);
        await db.into(db.variantInventoryMemory).insert(VariantInventoryMemoryCompanion.insert(
          variantId: variant.id, inventoryItemId: _inventoryItemId, sourceEventId: edit,
        ));
        await dao.actualizarProducto(product.id, ProductsCompanion(lastEventId: const Value(edit)));
        expect(await dao.restaurarRespaldoActualizacion(edit, product.lastEventId!), isTrue);
        expect(await db.select(db.variantInventoryMemory).get(), isEmpty);
      });
    }
    });
  }
}
const _inventoryItemId = '20000000-0000-4000-8000-000000000001';
const _missingInventoryItemId = '20000000-0000-4000-8000-000000000099';

List<LocalEventAppend> _entries({required String productInventoryItemId}) {
  final createdAt = DateTime.utc(2026, 8, 26);
  final resourceEvent = SyncEvent(
    eventId: '40000000-0000-4000-8000-000000000001',
    aggregateType: RecursoInventarioCreadoPayload.aggregateType,
    aggregateId: _inventoryItemId,
    eventType: RecursoInventarioCreadoPayload.eventType,
    deviceId: 'device-test',
    userId: 'user-test',
    baseVersion: 1,
    createdAtLocal: createdAt,
    payload: RecursoInventarioCreadoPayload.create(
      inventoryItemId: _inventoryItemId,
      name: 'Agua mineral',
      defaultUnitId: InventoryUnitIds.piece,
      initialMovement: InitialInventoryMovementPayload.create(
        movementId: '30000000-0000-4000-8000-000000000001',
        movementType: TipoMovimientoInventario.initialBalance,
        quantityDeltaAtomic: 15,
      ),
    ).toJson(),
  );
  final productEvent = SyncEvent(
    eventId: '40000000-0000-4000-8000-000000000002',
    aggregateType: ProductoCreadoPayload.aggregateType,
    aggregateId: '10000000-0000-4000-8000-000000000001',
    eventType: ProductoCreadoPayload.eventType,
    deviceId: 'device-test',
    userId: 'user-test',
    baseVersion: 1,
    createdAtLocal: createdAt,
    payload: ProductoCreadoPayload.create(
      nombre: 'Agua mineral',
      categoriaId: null,
      saleConfiguration: const UnitSaleConfiguration(),
      variantes: [
        ProductoCreadoVariante.create(
          id: '10000000-0000-4000-8000-000000000002',
          nombre: null,
          precioVentaMenor: 2500,
          costoEstandarMenor: null,
          inventoryItemId: productInventoryItemId,
          orden: 0,
        ),
      ],
      dependenciasInventario: [
        ProductoCreadoInventarioDependencia(
          refId: productInventoryItemId,
          dependsOnEventId: resourceEvent.eventId,
        ),
      ],
    ).toJson(),
  );
  return [
    LocalEventAppend(
      event: resourceEvent,
      refs: const [
        LocalEventRef.affects(
          refType: 'inventory_item',
          refId: _inventoryItemId,
        ),
      ],
    ),
    LocalEventAppend(
      event: productEvent,
      refs: [
        const LocalEventRef.affects(
          refType: 'product',
          refId: '10000000-0000-4000-8000-000000000001',
        ),
        LocalEventRef.uses(
          refType: 'inventory_item',
          refId: productInventoryItemId,
        ),
      ],
    ),
  ];
}
